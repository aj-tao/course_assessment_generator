\begin{tcolorbox}[colback=white,colframe=black,opacityframe=0,colbacktitle=black,title={\textbf{Problem 3 [\examproblemweight\ points]}}]
Consider the following functions 
$$f(x)=7-9x^2, \quad \quad
g(x) = 
\begin{cases}
	-\f2{x+2}+3 &\text{if}~x<-2\\
	8-3|x-2|&\text{if}~-2\le x<3\\
	5&\text{if}~x>3\\
\end{cases}, 
\quad \quad
h(x) = \f{x\sqrt{5+2x} + |x|}{x^2-4}.$$
\begin{enumerate}[label=(\alph*)]
	\item \textbf{(2 points)} Compute $g(-2)$.
	\item \textbf{(4 points)} Compute $\f{f(3+h)-f(3)}{h}$. (Leave your answer in terms of $h$, and try to simplify as much as possible.)
	\item \textbf{(4 points)} Compute the domain of $h$. Write your answer as an interval or as a union of intervals.
\end{enumerate}
\end{tcolorbox}
